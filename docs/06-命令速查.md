# 06 命令速查表

> 一页搞定日常操作。把 `<IP>`、`<密码>` 替换成你自己的。

## 安装 / 服务

```bash
# 依赖（含 ASF 必需的 libicu）
apt-get install -y curl unzip python3 ca-certificates
PKG=$(apt-cache search --names-only '^libicu[0-9]+$' | awk '{print $1}' | sort -V | tail -1) && apt-get install -y "$PKG"

# 解压 ASF（绕开老 unzip 的 zip64 限制）
python3 -c "import zipfile; zipfile.ZipFile('ASF-linux-x64.zip').extractall('/opt/asf')"
chmod +x /opt/asf/ArchiSteamFarm

# 服务
systemctl enable --now asf
systemctl restart asf
systemctl status asf --no-pager
systemctl is-active asf

# 日志
journalctl -u asf -f --no-pager
journalctl -u asf -n 100 --no-pager | grep -iE 'badge|farming|error|2FA'
journalctl --vacuum-time=7d
```

## 网络诊断（判断 DNS 污染 / SNI 阻断）

```bash
# 一键诊断
bash scripts/diagnose-steam-net.sh

# 解析（看是否有污染）
getent hosts steamcommunity.com
curl -sS 'https://dns.alidns.com/resolve?name=steamcommunity.com&type=A'

# 带 SNI（预期失败/超时）
curl -sS -o /dev/null -w '%{http_code} %{time_total}s\n' --max-time 15 https://steamcommunity.com/

# 不带 SNI（预期 200/302 且含 Steam 内容）
curl -sS -k -o /tmp/p.out -w '%{http_code}\n' --max-time 15 https://104.89.103.51/ -H 'Host: steamcommunity.com'
grep -o 'steam' /tmp/p.out | head -1

# 通过本地反代的最终验证
curl -sS -o /dev/null -w '首页 %{http_code}\n'  https://steamcommunity.com/
curl -sS -o /dev/null -w '徽章 %{http_code}\n'  https://steamcommunity.com/my/badges   # 302 = 正常

# CM（登录通道）连通性
python3 -c "
import socket
for h in ['103.28.54.162','146.66.152.38','155.133.248.38']:
    s=socket.socket(); s.settimeout(5)
    try: s.connect((h,27017)); print(h,'OK')
    except Exception as e: print(h,'FAIL',e)
    finally: s.close()"
```

## steam-proxy（社区反代）

```bash
# 一键部署（自动挑上游 IP）
bash scripts/setup-steam-proxy.sh

# 指定上游 IP 重新部署
UPSTREAM_IP=23.212.62.70 bash scripts/setup-steam-proxy.sh && systemctl restart steam-proxy

# 服务与端口
systemctl restart steam-proxy
systemctl is-active steam-proxy
ss -tlnp | grep -E '127.0.0.1:(443|8443)'
journalctl -u steam-proxy -n 30 --no-pager -o cat

# 证书（查看 SAN / 有效期）
openssl x509 -in /opt/steam-proxy/steamcommunity.crt -noout -text | grep -A2 'Subject Alternative'
openssl x509 -in /opt/steam-proxy/steamcommunity.crt -noout -dates

# CA 装进系统信任链
cp /opt/steam-proxy/cacert.crt /usr/local/share/ca-certificates/steam-proxy-ca.crt
update-ca-certificates
```

## hosts / iptables

```bash
# hosts
cat /etc/hosts | grep steam
sed -i '/steamcommunity.com/d' /etc/hosts            # 删除相关行

# iptables（仅 8443 方案需要）
iptables -t nat -L OUTPUT -n --line-numbers
iptables -t nat -D OUTPUT -p tcp -d 127.0.0.1 --dport 443 -j REDIRECT --to-ports 8443
```

## 账号 / 机器人

```bash
ls /opt/asf/config/                       # 机器人列表（文件名=机器人名）
cat /opt/asf/config/<机器人名>.json

# 提交 2FA 验证码（通过 IPC）
curl -sS -X POST http://127.0.0.1:1242/api/command \
  -H 'Authentication: <IPC密码>' -H 'Content-Type: application/json' \
  -d '{"Command":"2fa <机器人名> 12345"}'

# 查看 ASF 信息 / 机器人状态
curl -sS -H 'Authentication: <IPC密码>' http://127.0.0.1:1242/api/asf
curl -sS -H 'Authentication: <IPC密码>' http://127.0.0.1:1242/api/bot/<机器人名>
```

## 迁移（换机器免令牌）

```bash
# 旧机
systemctl stop asf
tar czf /tmp/asf-state.tgz -C /opt/asf config plugins
# 新机
tar xzf /tmp/asf-state.tgz -C /opt/asf && systemctl start asf
```

